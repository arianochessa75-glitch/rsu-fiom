    // Gestione sicura dei PDF tramite Blob locale
    function handleFileSelect(files) {
        if (files.length === 0) return;
        const file = files[0];
        
        if (file.type !== "application/pdf") {
            alert("Per favore seleziona un file in formato PDF valido.");
            return;
        }

        const fileURL = URL.createObjectURL(file);
        
        document.getElementById('fileNameDisplay').innerText = file.name;
        const openBtn = document.getElementById('openPdfBtn');
        const downloadBtn = document.getElementById('downloadPdfBtn');
        
        openBtn.href = fileURL;
        downloadBtn.href = fileURL;
        downloadBtn.download = file.name;
        
        document.getElementById('pdfViewerContainer').style.display = 'block';
    }

    // Drag and drop handler
    const dropZone = document.getElementById('dropZone');
    dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.style.background = '#ffe0e0'; });
    dropZone.addEventListener('dragleave', (e) => { e.preventDefault(); dropZone.style.background = '#fff5f5'; });
    dropZone.addEventListener('drop', (e) => {
        e.preventDefault();
        dropZone.style.background = '#fff5f5';
        if (e.dataTransfer.files.length > 0) {
            handleFileSelect(e.dataTransfer.files);
        }
    });

    // Calcolatore Giorni Lavorativi con Festività Nazionali e CCNL
    function calculateWorkingDays() {
        const startStr = document.getElementById('startDate').value;
        const endStr = document.getElementById('endDate').value;
        
        if (!startStr || !endStr) {
            alert("Inserisci sia la data di inizio che la data di fine.");
            return;
        }

        let startDate = new Date(startStr);
        let endDate = new Date(endStr);

        if (startDate > endDate) {
            alert("La data di inizio deve essere precedente o uguale alla data di fine.");
            return;
        }

        let workingDays = 0;
        let holidaysFound = [];
        let currentDate = new Date(startDate);

        while (currentDate <= endDate) {
            let dayOfWeek = currentDate.getDay(); // 0 = Domenica, 6 = Sabato
            let year = currentDate.getFullYear();
            let month = currentDate.getMonth(); 
            let day = currentDate.getDate();

            let holidayName = null;

            // Festività Nazionali Fisse
            const fixedHolidays = [
                {m: 0, d: 1, name: "Capodanno (1° Gennaio)"},
                {m: 0, d: 6, name: "Epifania (6 Gennaio)"},
                {m: 3, d: 25, name: "Festa della Liberazione (25 Aprile)"},
                {m: 4, d: 1, name: "Festa dei Lavoratori (1° Maggio)"},
                {m: 5, d: 2, name: "Festa della Repubblica (2 Giugno)"},
                {m: 7, d: 15, name: "Ferragosto (15 Agosto)"},
                {m: 9, d: 4, name: "Giornata dell'Unità Nazionale (4 Novembre)"},
                {m: 10, d: 1, name: "Tutti i Santi (1° Novembre)"},
                {m: 11, d: 8, name: "Immacolata Concezione (8 Dicembre)"},
                {m: 11, d: 25, name: "Natale (25 Dicembre)"},
                {m: 11, d: 26, name: "Santo Stefano (26 Dicembre)"}
            ];

            for (let h of fixedHolidays) {
                if (month === h.m && day === h.d) {
                    holidayName = h.name;
                    break;
                }
            }

            // Calcolo Pasqua e Pasquetta (Algoritmo di Gauss)
            let a = year % 19,
                b = Math.floor(year / 100),
                c = year % 100,
                d = Math.floor(b / 4),
                e = b % 4,
                f = Math.floor((b + 8) / 25),
                g = Math.floor((b - f + 1) / 3),
                h_g = (19 * a + b - d - g + 15) % 30,
                i = Math.floor(c / 4),
                k = c % 4,
                L = (32 + 2 * e + 2 * i - h_g - k) % 7,
                m = Math.floor((a + 11 * h_g + 22 * L) / 451),
                monthPasqua = Math.floor((h_g + L - 7 * m + 114) / 31) - 1,
                dayPasqua = ((h_g + L - 7 * m + 114) % 31) + 1;
            
            let datePasqua = new Date(year, monthPasqua, dayPasqua);
            let datePasquetta = new Date(year, monthPasqua, dayPasqua + 1);

            if (currentDate.getTime() === datePasqua.getTime()) {
                holidayName = "Pasqua";
            } else if (currentDate.getTime() === datePasquetta.getTime()) {
                holidayName = "Lunedì dell'Angelo / Pasquetta";
            }

            // Verifica se è festivo o weekend
            if (holidayName) {
                let note = holidayName;
                if (dayOfWeek === 0) {
                    note += " (Cadendo di domenica, dà diritto al trattamento economico aggiuntivo come previsto da CCNL)";
                }
                holidaysFound.push({
                    date: new Date(currentDate).toLocaleDateString('it-IT'),
                    description: note
                });
            }

            // Conta se non è sabato (6), domenica (0) e non è festivo
            if (dayOfWeek !== 0 && dayOfWeek !== 6 && !holidayName) {
                workingDays++;
            }

            currentDate.setDate(currentDate.getDate() + 1);
        }

        const resultDiv = document.getElementById('calcResult');
        resultDiv.style.display = 'block';
        
        let htmlContent = `<strong>📊 Risultato del Calcolo:</strong><br>`;
        htmlContent += `Giorni lavorativi effettivi: <strong>${workingDays}</strong><br>`;
        htmlContent += `Festività nazionali/contrattuali nel periodo: <strong>${holidaysFound.length}</strong>`;
        
        if (holidaysFound.length > 0) {
            htmlContent += `<ul>`;
            for (let hol of holidaysFound) {
                htmlContent += `<li><strong>${hol.date}</strong>: ${hol.description}</li>`;
            }
            htmlContent += `</ul>`;
        }
        
        resultDiv.innerHTML = htmlContent;
    }
